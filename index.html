import os import requests import gradio as gr import spaces DEEPSEEK_KEY = os.getenv("LLM_API_KEY") API_URL = "https://api.deepseek.com/v1/chat/completions" SYSTEM_PROMPT = """ 你是一名金融、高数备考助教，面向南开大学金融科技大一学生。 擅长：公司金融、货币银行学、投资学、商业银行管理、高数、数竞非数B。 要求：人话讲解，步骤清晰，重点标注易错点；计算题给简洁步骤；生成单选严格输出JSON，不要多余文字。 """ @spaces.GPU(duration=120) def chat_ask(question): if not DEEPSEEK_KEY: return "❌ 环境变量 LLM_API_KEY 未配置" if not question.strip(): return "⚠️ 请输入问题" headers = { "Authorization": f"Bearer {DEEPSEEK_KEY}", "Content-Type": "application/json" } payload = { "model": "deepseek-chat", "messages": [ {"role": "system", "content": SYSTEM_PROMPT}, {"role": "user", "content": question} ], "temperature": 0.3 } try: resp = requests.post(API_URL, headers=headers, json=payload, timeout=30) resp.raise_for_status() return resp.json()["choices"][0]["message"]["content"] except Exception as e: return f"❌ 接口异常：{str(e)}" demo = gr.Interface( fn=chat_ask, inputs=gr.Textbox(visible=False), outputs=gr.Textbox(visible=False), ).queue() if __name__ == "__main__": demo.launch(server_name="0.0.0.0", show_error=True)
